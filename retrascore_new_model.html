1 Analysis design

Model development is performed exclusively with patients from 2015-2019. The 2020-2025 cohort is kept separate and is used only for temporal validation. No variable selection, coefficient estimation, point assignment, or calibration is performed with the 2020-2025 outcomes.

Following the Datanovia stepwise-logistic-regression workflow, the modelling procedure follows these steps:

  1. Fit a full binomial logistic regression containing every predictor explicitly listed in the editable candidate-variable block below.
  2. Apply bidirectional stepwise selection with MASS::stepAIC(), minimizing Akaike’s information criterion (AIC).
  3. Use the refitted model returned by stepAIC() as the new multivariable logistic model.
  4. Multiply every non-intercept coefficient by 2 and round it to the nearest integer, following the article.
  5. Sum those same fixed points for every patient in both cohorts.
  6. Estimate the score-to-mortality equation in 2015-2019 and apply it unchanged to both cohorts.
  7. Compare observed and expected 30-day mortality in 2015-2019 and 2020-2025.

Variable selection and coefficient estimation use only the 2015-2019 cohort. The 2020-2025 outcomes are not examined by the stepwise procedure and no logistic regression is fitted in the temporal-validation cohort.

3 Stepwise logistic regression

full_model_formula <- reformulate(
  candidate_variables,
  response = "mortalidad_30d"
)
full_model <- glm(
  full_model_formula,
  data = derivation,
  family = binomial(link = "logit"),
  na.action = na.fail
)
if (!full_model$converged) stop("The full logistic regression did not converge.")
if (anyNA(coef(full_model))) {
  stop("The full logistic regression contains non-estimable coefficients.")
}

# Both-direction selection starts from the full model. At each step, stepAIC()
# considers adding or removing a term and retains a change only when it lowers
# AIC. The returned object is already refitted and is the new model used below.
stepwise_model <- MASS::stepAIC(
  full_model,
  direction = "both",
  trace = FALSE
)
if (!stepwise_model$converged) {
  stop("The stepwise-selected logistic regression did not converge.")
}
if (anyNA(coef(stepwise_model))) {
  stop("The stepwise-selected model contains non-estimable coefficients.")
}

selected_variables <- attr(terms(stepwise_model), "term.labels")
if (!length(selected_variables)) {
  stop("Stepwise selection produced an intercept-only model.")
}

stepwise_path <- stepwise_model$anova
names(stepwise_path) <- make.names(names(stepwise_path), unique = TRUE)
selection_summary <- data.frame(

  Full_model_AIC = AIC(full_model),

3.1 Variable-inclusion plot

This plot is the logistic-regression analogue of a leaps::regsubsets() plot. Rows are complete model terms, including categorical variables as single terms; columns are the successive models visited by stepAIC(). Because the outcome is binary, model quality is shown with AIC rather than adjusted R-squared.

step_actions <- trimws(as.character(stepwise_path$Step))
step_actions[is.na(step_actions)] <- ""

included_at_step <- vector("list", nrow(stepwise_path))
current_variables <- candidate_variables
included_at_step[[1]] <- current_variables

if (nrow(stepwise_path) > 1L) {
  for (step_index in 2:nrow(stepwise_path)) {
    action <- step_actions[step_index]
    variable_name <- trimws(sub("^[+-]\\s*", "", action))
    if (startsWith(action, "-")) {
      current_variables <- setdiff(current_variables, variable_name)
    } else if (startsWith(action, "+")) {
      current_variables <- union(current_variables, variable_name)
    }
    included_at_step[[step_index]] <- current_variables
  }
}

step_labels <- ifelse(
  seq_len(nrow(stepwise_path)) == 1L,
  "Initial full model",
  step_actions
)
aic_labels <- paste0(
  seq_len(nrow(stepwise_path)) - 1L,
  "\nAIC = ", formatC(stepwise_path$AIC, digits = 1, format = "f")
)

stepwise_inclusion <- do.call(
  rbind,
  lapply(seq_len(nrow(stepwise_path)), function(step_index) {
    data.frame(
      Step_index = step_index - 1L,
      Step = step_labels[step_index],
      AIC = stepwise_path$AIC[step_index],
      Variable = candidate_variables,
      Included = candidate_variables %in% included_at_step[[step_index]],
      stringsAsFactors = FALSE
    )
  })
)
stepwise_inclusion$Variable <- factor(
  stepwise_inclusion$Variable,
  levels = rev(candidate_variables)
)

5 Fixed integer score

Following the article, each non-intercept regression coefficient is multiplied by 2 and rounded to the nearest integer. This creates one fixed point system. The same points are applied to patients in both cohorts.

round_nearest_integer <- function(x) {
  sign(x) * floor(abs(x) + 0.5)
}

regression_betas <- coef(logistic_model)[-1]
score_points <- setNames(
  as.integer(round_nearest_integer(2 * regression_betas)),
  names(regression_betas)
)
points_table <- data.frame(
  Term = names(regression_betas),
  Beta = as.numeric(regression_betas),
  `Beta multiplied by 2` = 2 * as.numeric(regression_betas),
  Points = as.integer(score_points),
  check.names = FALSE,
  row.names = NULL
)
score_formula <- reformulate(selected_variables)

reference_rows <- lapply(names(logistic_model$xlevels), function(variable_name) {

    Term = paste0(
      variable_name, logistic_model$xlevels[[variable_name]][1], " (reference)"

calculate_score <- function(data) {
  # Use a predictor-only formula so rows with an unknown outcome are not
  # removed or shifted while assigning patient-level scores.
  design_matrix <- model.matrix(score_formula, data = data)[, -1, drop = FALSE]
  missing_columns <- setdiff(names(score_points), colnames(design_matrix))
  if (length(missing_columns)) {
    stop("Missing score columns: ", paste(missing_columns, collapse = ", "))
  }
  design_matrix <- design_matrix[, names(score_points), drop = FALSE]
  as.integer(drop(design_matrix %*% score_points))
}

# Score every patient with all selected predictors, whether or not the outcome
# is known. Missing predictors remain explicit and are never assigned zero.
score_available <- complete.cases(
  eligible[, selected_variables, drop = FALSE]
)
eligible$new_score <- NA_integer_
eligible$new_score[score_available] <- calculate_score(
  eligible[score_available, , drop = FALSE]
)

# Estimate the score-to-mortality equation only in the derivation cases used to
# fit the logistic regression, then freeze it for both cohorts.
derivation$new_score <- calculate_score(derivation)
score_probability_model <- glm(
  mortalidad_30d ~ new_score,
  data = derivation,
  family = binomial(link = "logit")
)
if (!score_probability_model$converged) {
  stop("The score-to-mortality model did not converge.")
}

eligible$expected_mortality <- NA_real_
eligible$expected_mortality[score_available] <- predict(
  score_probability_model,
  newdata = data.frame(new_score = eligible$new_score[score_available]),
  type = "response"
)

summarise_observed_expected <- function(data, cohort_name) {
  deaths <- sum(data$mortalidad_30d)
  patients <- nrow(data)
  observed_ci <- stats::binom.test(deaths, patients)$conf.int
  data.frame(
    Cohort = cohort_name,
    Patients = patients,
    Deaths = deaths,
    `Observed mortality (%)` = 100 * deaths / patients,
    `Observed 95% CI lower` = 100 * observed_ci[1],
    `Observed 95% CI upper` = 100 * observed_ci[2],
    `Expected deaths` = sum(data$expected_mortality),
    `Expected mortality (%)` = 100 * mean(data$expected_mortality),

7 Score performance

evaluate_score <- function(data, cohort_name) {
  outcome <- data$mortalidad_30d
  probability <- data$expected_mortality
  roc_object <- pROC::roc(
    outcome, probability, levels = c(0, 1), direction = "<", quiet = TRUE
  )
  auc_ci <- as.numeric(pROC::ci.auc(roc_object, method = "delong"))
  lp <- qlogis(pmin(pmax(probability, 1e-12), 1 - 1e-12))
  intercept_model <- glm(outcome ~ 1, offset = lp, family = binomial())
  slope_model <- glm(outcome ~ lp, family = binomial())
  data.frame(
    Cohort = cohort_name,
    N = length(outcome),
    AUROC = as.numeric(pROC::auc(roc_object)),
    `AUROC 95% CI lower` = auc_ci[1],
    `AUROC 95% CI upper` = auc_ci[3],
    Brier = mean((outcome - probability)^2),
    `Calibration intercept` = unname(coef(intercept_model)[1]),
    `Calibration slope` = unname(coef(slope_model)[2]),
    check.names = FALSE
  )
}

9 Reproducibility notes

  • Missing candidate values are handled by complete-case analysis; no value is imputed and no missing predictor is interpreted as zero.
  • All 20 explicitly listed predictor variables enter the initial full logistic model. The bidirectional stepAIC() search removes or re-adds complete model terms, including categorical predictors as whole terms.
  • All candidate models are fitted to the same complete-case derivation sample, making their AIC values comparable. The 2020-2025 results do not influence variable selection.
  • Integer points are based on the stepwise-selected logistic-regression coefficients: points = round(2 x beta).
  • The selected logistic regression is fitted in 2015-2019. Its fixed point table and score-to-probability equation are applied unchanged to both cohorts.
  • No model coefficient, point value, or probability equation is re-estimated with the 2020-2025 cohort.
  • Post-selection confidence intervals and P values do not account for the uncertainty introduced by stepwise selection and should be interpreted descriptively. Stepwise selection can also be unstable under resampling.
